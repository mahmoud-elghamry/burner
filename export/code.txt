  VAR_INPUT
    Auto                               : Bool
    StartPB                            : Bool
    StopOK                             : Bool
    LevelOK                            : Bool
    RunFb                              : Bool
    Lockout                            : Bool
    Fuel                               : Bool
    TeA                                : Int
    TeB                                : Int
    Use24h                             : Bool
    EStopOK                            : Bool
    ResetPulse                         : Bool

    SP                                 : Int
    HYST                               : Int
    STG                                : Int
    TStart                             : Time

  VAR_OUTPUT
    HeatReq                            : Bool
    Stage2                             : Bool
    State                              : Int
    Enabled                            : Bool
    Tmax                               : Int
    F_EStop                            : Bool
    F_TC                               : Bool
    F_Lockout                          : Bool
    F_NoStart                          : Bool
    F_RunLost                          : Bool
    F_RunWrong                         : Bool

  VAR_STATIC
    StartEdge                          : Bool
    StartMem                           : Bool
    AutoChg                            : Bool
    AutoMem                            : Bool
    FuelChg                            : Bool
    FuelMem                            : Bool
    TcOkA                              : Bool
    TcOkB                              : Bool
    TcOk                               : Bool
    TcBad                              : Bool
    Demand                             : Bool
    HeatNeed                           : Bool
    Ready                              : Bool
    Active                             : Bool
    Pause24                            : Bool
    RunWrong                           : Bool
    NoStart                            : Bool
    SP_Act                             : Int
    HYST_Act                           : Int
    STG_Act                            : Int
    TStart_Act                         : Time
    TOn                                : Int
    TStg                               : Int
    T_TcBad                            : TON_TIME

    T_RunOff                           : TON_TIME

  VAR_NONE
    PT                                 : Time

------------------------------------------------------------------------------
NETWORK 1  --  R-05 — START edge: not on first scan, only after seen released, STOP wins
------------------------------------------------------------------------------
   #StartEdge := #StartPB AND NOT #StartMem AND NOT #FirstScan AND #StopOK

------------------------------------------------------------------------------
NETWORK 2  --  R-05 — START edge memory
------------------------------------------------------------------------------
   #StartMem := #StartPB

------------------------------------------------------------------------------
NETWORK 3  --  R-02 — AUTO/MAN changed
------------------------------------------------------------------------------
   #AutoChg := (#Auto AND NOT #AutoMem OR NOT #Auto AND #AutoMem)

------------------------------------------------------------------------------
NETWORK 4  --  R-02 — AUTO/MAN memory
------------------------------------------------------------------------------
   #AutoMem := #Auto

------------------------------------------------------------------------------
NETWORK 5  --  R-12 — fuel mode changed
------------------------------------------------------------------------------
   #FuelChg := (#Fuel AND NOT #FuelMem OR NOT #Fuel AND #FuelMem)

------------------------------------------------------------------------------
NETWORK 6  --  R-12 — fuel mode memory
------------------------------------------------------------------------------
   #FuelMem := #Fuel

------------------------------------------------------------------------------
NETWORK 7  --  C0xd — channel A valid: -20.0 .. 150.0 degC (32767 / -32768 fall outside)
------------------------------------------------------------------------------
   #TcOkA := (#TeA >= -200) AND (#TeA <= 1500)

------------------------------------------------------------------------------
NETWORK 8  --  C0xd — channel B valid
------------------------------------------------------------------------------
   #TcOkB := (#TeB >= -200) AND (#TeB <= 1500)

------------------------------------------------------------------------------
NETWORK 9  --  C0xd — both channels valid, each checked before MAX
------------------------------------------------------------------------------
   #TcOk := #TcOkA AND #TcOkB

------------------------------------------------------------------------------
NETWORK 10  --  C0xd / R-10 — invalid for more than 10 s = fault (heat is blocked at once)
------------------------------------------------------------------------------
   #TcBad := TON(IN:=NOT #TcOk, PT:=T#10S)[T_TcBad].Q

------------------------------------------------------------------------------
NETWORK 11  --  C0xb — temperature = higher channel (A)
------------------------------------------------------------------------------
   IF (#TeA >= #TeB) THEN  MOVE  #TeA  ->  #Tmax

------------------------------------------------------------------------------
NETWORK 12  --  C0xb — temperature = higher channel (B)
------------------------------------------------------------------------------
   IF (#TeA < #TeB) THEN  MOVE  #TeB  ->  #Tmax

------------------------------------------------------------------------------
NETWORK 13  --  R-11 — burner not heating: take all parameters
------------------------------------------------------------------------------
   IF NOT #Active THEN  MOVE  #SP  ->  #SP_Act
   IF NOT #Active THEN  MOVE  #STG  ->  #STG_Act
   IF NOT #Active THEN  MOVE  #TStart  ->  #TStart_Act

------------------------------------------------------------------------------
NETWORK 14  --  R-11 — lower SP applies at once
------------------------------------------------------------------------------
   IF (#SP <= #SP_Act) THEN  MOVE  #SP  ->  #SP_Act

------------------------------------------------------------------------------
NETWORK 15  --  R-11 — higher STG (less 2nd stage) applies at once
------------------------------------------------------------------------------
   IF (#STG >= #STG_Act) THEN  MOVE  #STG  ->  #STG_Act

------------------------------------------------------------------------------
NETWORK 16  --  R-11 — HYST only acts while stopped, so it is taken at once
------------------------------------------------------------------------------
   MOVE  #HYST  ->  #HYST_Act

------------------------------------------------------------------------------
NETWORK 17  --  C0xb / C0xc — restart and 2nd-stage limits
------------------------------------------------------------------------------
   (no statements decoded)

------------------------------------------------------------------------------
NETWORK 18  --  C0xb — AUTO demand on at T <= SP - HYST
------------------------------------------------------------------------------
   #Demand := S (#Tmax <= #TOn)

------------------------------------------------------------------------------
NETWORK 19  --  C0xb — AUTO demand off at T >= SP, or no valid temperature
------------------------------------------------------------------------------
   #Demand := R ((#Tmax >= #SP_Act) OR NOT #TcOk)

------------------------------------------------------------------------------
NETWORK 20  --  B-01 — heat needed: AUTO = hysteresis, MAN = below SP
------------------------------------------------------------------------------
   #HeatNeed := (#TcOk AND #Auto AND #Demand OR #TcOk AND NOT #Auto AND (#Tmax < #SP_Act))

------------------------------------------------------------------------------
NETWORK 21  --  C0xa / C0xd / C01 / C0xe / C07h — ready to fire
------------------------------------------------------------------------------
   #Ready := #LevelOK AND #TcOk AND #EStopOK AND NOT #Lockout AND NOT #Pause24

------------------------------------------------------------------------------
NETWORK 22  --  C07h / R-08 — 24 h of continuous request: forced stop
------------------------------------------------------------------------------
   #Pause24 := S TON(IN:=#Use24h AND #Active, PT:=T#24H)[T_24h].Q

------------------------------------------------------------------------------
NETWORK 23  --  R-08 — restart allowed after RUN has been off 60 s
------------------------------------------------------------------------------
   #Pause24 := R TON(IN:=#Pause24 AND NOT #RunFb, PT:=T#60S)[T_Pause].Q

------------------------------------------------------------------------------
NETWORK 24  --  R-09 — RUN on without our request for 30 s
------------------------------------------------------------------------------
   #RunWrong := TON(IN:=#RunFb AND NOT #Active, PT:=T#30S)[T_RunOff].Q

------------------------------------------------------------------------------
NETWORK 25  --  C0xf — no RUN within T_START
------------------------------------------------------------------------------
   #NoStart := TON(IN:=(#State = 1), PT:=#TStart_Act)[T_Start].Q

------------------------------------------------------------------------------
NETWORK 26  --  R-01 / B-01 — START accepted: IDLE, ready, and in MAN only if heat is needed
------------------------------------------------------------------------------
   #Enabled := S (#StartEdge AND (#State = 0) AND #Ready AND #Auto OR #StartEdge AND (#State = 0) AND #Ready AND #HeatNeed)

------------------------------------------------------------------------------
NETWORK 27  --  B-01 / R-02 / R-12 — STOP, AUTO/MAN change or fuel change take the START back
------------------------------------------------------------------------------
   #Enabled := R (NOT #StopOK OR #AutoChg OR #FuelChg)

------------------------------------------------------------------------------
NETWORK 28  --  B-01 — IDLE -> STARTING (R-09: not if RUN is already on)
------------------------------------------------------------------------------
   IF (#State = 0) AND #Enabled AND #Ready AND #HeatNeed AND NOT #RunFb THEN  MOVE  1  ->  #State

------------------------------------------------------------------------------
NETWORK 29  --  B-01 — STARTING -> RUNNING on RUN
------------------------------------------------------------------------------
   IF (#State = 1) AND #RunFb THEN  MOVE  2  ->  #State

------------------------------------------------------------------------------
NETWORK 30  --  C0xa / C0xb / C07h / B-01 — STARTING or RUNNING -> IDLE
------------------------------------------------------------------------------
   IF (((#State = 1) OR (#State = 2)) AND NOT #Ready OR ((#State = 1) OR (#State = 2)) AND NOT #HeatNeed OR ((#State = 1) OR (#State = 2)) AND NOT #Enabled) THEN  MOVE  0  ->  #State

------------------------------------------------------------------------------
NETWORK 31  --  R-03 — MAN: one cycle; any stop needs a new START
------------------------------------------------------------------------------
   #Enabled := R NOT #Auto AND (#State = 0) AND #Active

------------------------------------------------------------------------------
NETWORK 32  --  C0xf — failed to start
------------------------------------------------------------------------------
   IF (#State = 1) AND #NoStart THEN  MOVE  3  ->  #State
   #F_NoStart := S (#State = 1) AND #NoStart

------------------------------------------------------------------------------
NETWORK 33  --  C0xg — RUN lost while running, no lockout, not stopped by us
------------------------------------------------------------------------------
   IF (#State = 2) AND NOT #RunFb AND NOT #Lockout THEN  MOVE  3  ->  #State
   #F_RunLost := S (#State = 2) AND NOT #RunFb AND NOT #Lockout

------------------------------------------------------------------------------
NETWORK 34  --  R-06 / R-07 — RESET leaves FAULT only when every cause is gone
------------------------------------------------------------------------------
   IF (#State = 3) AND #ResetPulse AND #EStopOK AND NOT #Lockout AND #TcOk AND NOT #RunFb THEN  MOVE  0  ->  #State
   #F_EStop := R (#State = 3) AND #ResetPulse AND #EStopOK AND NOT #Lockout AND #TcOk AND NOT #RunFb
   #F_TC := R (#State = 3) AND #ResetPulse AND #EStopOK AND NOT #Lockout AND #TcOk AND NOT #RunFb
   #F_Lockout := R (#State = 3) AND #ResetPulse AND #EStopOK AND NOT #Lockout AND #TcOk AND NOT #RunFb
   #F_NoStart := R (#State = 3) AND #ResetPulse AND #EStopOK AND NOT #Lockout AND #TcOk AND NOT #RunFb
   #F_RunLost := R (#State = 3) AND #ResetPulse AND #EStopOK AND NOT #Lockout AND #TcOk AND NOT #RunFb
   #F_RunWrong := R (#State = 3) AND #ResetPulse AND #EStopOK AND NOT #Lockout AND #TcOk AND NOT #RunFb

------------------------------------------------------------------------------
NETWORK 35  --  C01 — E-STOP
------------------------------------------------------------------------------
   IF NOT #EStopOK THEN  MOVE  3  ->  #State
   #F_EStop := S NOT #EStopOK

------------------------------------------------------------------------------
NETWORK 36  --  C0xe — burner lockout
------------------------------------------------------------------------------
   IF #Lockout THEN  MOVE  3  ->  #State
   #F_Lockout := S #Lockout

------------------------------------------------------------------------------
NETWORK 37  --  C0xd — thermocouple fault
------------------------------------------------------------------------------
   IF #TcBad THEN  MOVE  3  ->  #State
   #F_TC := S #TcBad

------------------------------------------------------------------------------
NETWORK 38  --  R-09 — RUN without request
------------------------------------------------------------------------------
   IF #RunWrong THEN  MOVE  3  ->  #State
   #F_RunWrong := S #RunWrong

------------------------------------------------------------------------------
NETWORK 39  --  B-02 — FAULT takes the START back
------------------------------------------------------------------------------
   #Enabled := R (#State = 3)

------------------------------------------------------------------------------
NETWORK 40  --  E1 / E3 — heat request while STARTING or RUNNING
------------------------------------------------------------------------------
   #Active := ((#State = 1) OR (#State = 2))
   #HeatReq := ((#State = 1) OR (#State = 2))

------------------------------------------------------------------------------
NETWORK 41  --  C0xc — 2nd stage: RUNNING and T < SP - STG
------------------------------------------------------------------------------
   #Stage2 := (#State = 2) AND (#Tmax < #TStg)

##############################################################################
# BLOCK FC   Common   (LAD, No.1, Optimized)
##############################################################################

  VAR_RETURN
    Ret_Val                            : Void

------------------------------------------------------------------------------
NETWORK 1  --  R-06 — RESET edge (not on first scan)
------------------------------------------------------------------------------
   "Sys.ResetPulse" := "S2_FaultReset" AND NOT "Sys.ResetMem" AND NOT "Sys.FirstScan"

------------------------------------------------------------------------------
NETWORK 2  --  R-06 — RESET edge memory
------------------------------------------------------------------------------
   "Sys.ResetMem" := "S2_FaultReset"

------------------------------------------------------------------------------
NETWORK 3  --  C01 — E-STOP pressed (NC1 open) drops the latch
------------------------------------------------------------------------------
   "Sys.EStopOK" := R NOT "S0_EStopStatus"

------------------------------------------------------------------------------
NETWORK 4  --  C01 — E-STOP latch back after RESET (or at power-up if released)
------------------------------------------------------------------------------
   "Sys.EStopOK" := S ("S0_EStopStatus" AND "Sys.ResetPulse" OR "S0_EStopStatus" AND "Sys.FirstScan")

------------------------------------------------------------------------------
NETWORK 5  --  C01 — E-STOP alarm
------------------------------------------------------------------------------
   "Sys.Alm_EStop" := NOT "Sys.EStopOK"

------------------------------------------------------------------------------
NETWORK 6  --  R-11 — RESET clears the parameter alarm
------------------------------------------------------------------------------
   "Sys.Alm_ParRejected" := R "Sys.ResetPulse"

------------------------------------------------------------------------------
NETWORK 7  --  R-11 — SP: stored value bad after power-up -> default
------------------------------------------------------------------------------
   IF ("Sys.FirstScan" AND ("Par.SP_V" < 400) OR "Sys.FirstScan" AND ("Par.SP_V" > 900)) THEN  MOVE  700  ->  "Par.SP_V"

------------------------------------------------------------------------------
NETWORK 8  --  R-11 — SP: entry in range -> used
------------------------------------------------------------------------------
   IF ("Par.SP" >= 400) AND ("Par.SP" <= 900) THEN  MOVE  "Par.SP"  ->  "Par.SP_V"

------------------------------------------------------------------------------
NETWORK 9  --  R-11 — SP: entry out of range -> old value back, alarm
------------------------------------------------------------------------------
   IF (("Par.SP" < 400) OR ("Par.SP" > 900)) THEN  MOVE  "Par.SP_V"  ->  "Par.SP"
   "Sys.Alm_ParRejected" := S (("Par.SP" < 400) OR ("Par.SP" > 900))

------------------------------------------------------------------------------
NETWORK 10  --  R-11 — HYST: stored value bad after power-up -> default
------------------------------------------------------------------------------
   IF ("Sys.FirstScan" AND ("Par.HYST_V" < 30) OR "Sys.FirstScan" AND ("Par.HYST_V" > 100)) THEN  MOVE  50  ->  "Par.HYST_V"

------------------------------------------------------------------------------
NETWORK 11  --  R-11 — HYST: entry in range -> used
------------------------------------------------------------------------------
   IF ("Par.HYST" >= 30) AND ("Par.HYST" <= 100) THEN  MOVE  "Par.HYST"  ->  "Par.HYST_V"

------------------------------------------------------------------------------
NETWORK 12  --  R-11 — HYST: entry out of range -> old value back, alarm
------------------------------------------------------------------------------
   IF (("Par.HYST" < 30) OR ("Par.HYST" > 100)) THEN  MOVE  "Par.HYST_V"  ->  "Par.HYST"
   "Sys.Alm_ParRejected" := S (("Par.HYST" < 30) OR ("Par.HYST" > 100))

------------------------------------------------------------------------------
NETWORK 13  --  R-11 — STG: stored value bad after power-up -> default
------------------------------------------------------------------------------
   IF ("Sys.FirstScan" AND ("Par.STG_V" < 50) OR "Sys.FirstScan" AND ("Par.STG_V" > 300)) THEN  MOVE  100  ->  "Par.STG_V"

------------------------------------------------------------------------------
NETWORK 14  --  R-11 — STG: entry in range -> used
------------------------------------------------------------------------------
   IF ("Par.STG" >= 50) AND ("Par.STG" <= 300) THEN  MOVE  "Par.STG"  ->  "Par.STG_V"

------------------------------------------------------------------------------
NETWORK 15  --  R-11 — STG: entry out of range -> old value back, alarm
------------------------------------------------------------------------------
   IF (("Par.STG" < 50) OR ("Par.STG" > 300)) THEN  MOVE  "Par.STG_V"  ->  "Par.STG"
   "Sys.Alm_ParRejected" := S (("Par.STG" < 50) OR ("Par.STG" > 300))

------------------------------------------------------------------------------
NETWORK 16  --  R-11 — TStart: stored value bad after power-up -> default
------------------------------------------------------------------------------
   IF ("Sys.FirstScan" AND ("Par.TStart_V" < T#90S) OR "Sys.FirstScan" AND ("Par.TStart_V" > T#5M)) THEN  MOVE  T#2M  ->  "Par.TStart_V"

------------------------------------------------------------------------------
NETWORK 17  --  R-11 — TStart: entry in range -> used
------------------------------------------------------------------------------
   IF ("Par.TStart" >= T#90S) AND ("Par.TStart" <= T#5M) THEN  MOVE  "Par.TStart"  ->  "Par.TStart_V"

------------------------------------------------------------------------------
NETWORK 18  --  R-11 — TStart: entry out of range -> old value back, alarm
------------------------------------------------------------------------------
   IF (("Par.TStart" < T#90S) OR ("Par.TStart" > T#5M)) THEN  MOVE  "Par.TStart_V"  ->  "Par.TStart"
   "Sys.Alm_ParRejected" := S (("Par.TStart" < T#90S) OR ("Par.TStart" > T#5M))
